#### ДЗ Пишем Bash скрипт
Написать скрипт для CRON, который раз в час формирует отчёт и отправляет его на заданную почту.


Отчёт должен содержать:

IP-адреса с наибольшим числом запросов (с момента последнего запуска);
Запрашиваемые URL с наибольшим числом запросов (с момента последнего запуска);
Ошибки веб-сервера/приложения (с момента последнего запуска);
HTTP-коды ответов с указанием их количества (с момента последнего запуска).

##### Скрипт в cron_report.sh, анализируются логи файла access-4560-644067.log
```
#!/bin/bash

# ==========================================
# НАСТРОЙКИ
# ==========================================
TO_EMAIL="some_admin@mail.ru"
LOG_DIR="/var/log/nginx"                  # Директория с логами
LOG_PATTERN="access-4560-644067.log"                 # Шаблон файлов логов
REPORT_FILE="/tmp/cron_hourly_report.txt"
FILTERED_LOG="/tmp/cron_filtered.log"
MARKER_FILE="/tmp/.report_last_run"      # Файл-маркер времени последнего запуска

# ==========================================
# ТРАП (ОБРАБОТКА СИГНАЛОВ И ОЧИСТКА)
# ==========================================
cleanup() {
    local exit_code=$?
    # Обновляем маркер времени ТОЛЬКО при успешном завершении скрипта
    if [ $exit_code -eq 0 ]; then
        touch "$MARKER_FILE"
    fi
    rm -f "$REPORT_FILE" "$FILTERED_LOG"
    exit $exit_code
}
trap cleanup EXIT INT TERM

# ==========================================
# ФУНКЦИИ
# ==========================================

# 1. Функция фильтрации новых строк с момента последнего запуска
filter_new_logs() {
    # Находим все файлы логов, которые были изменены позже, чем создался наш маркер
    local new_logs=()
    
    if [ -f "$MARKER_FILE" ]; then
        # Читаем только файлы, изменившиеся с момента последнего запуска
        while IFS= read -r file; do
            new_logs+=("$file")
        done < <(find "$LOG_ ScamDIR" -maxdepth 1 -type f -name "$LOG_PATTERN" -newer "$MARKER_FILE")
    else
        # Если это самый первый запуск — берем последний измененный лог-файл целиком
        local initial_log
        initial_log=$(find "$LOG_DIR" -maxdepth 1 -type f -name "$LOG_PATTERN" -printf "%T@ %p\n" 2>/dev/null | sort -n | tail -n 1 | awk '{print $2}')
        if [ -n "$initial_log" ]; then
            new_logs+=("$initial_log")
        fi
    fi

    if [ ${#new_logs[@]} -eq 0 ]; then
        return 1 # Нет новых или изменившихся файлов логов
    fi

    # Сбрасываем старый отфильтрованный лог
    > "$FILTERED_LOG"

    # Если маркер существует, вытаскиваем точное время последнего запуска в формате лога
    # Пример: 11/Nov/2026:12:00:00 (учитывая секунды)
    if [ -f "$MARKER_FILE" ]; then
        local last_run_time
        last_run_time=$(date -r "$MARKER_FILE" +"%d/%b/%Y:%H:%M:%S")
        
        # Экранируем спецсимволы для sed
        local escaped_time
        escaped_time=$(echo "$last_run_time" | sed 's/\//\\\//g; s/:/\\:/g')

        # С помощью sed вырезаем все строки от момента последней отметки времени и до конца файла
        for log in "${new_logs[@]}"; do
            sed -n "/$escaped_time/,\$p" "$log" >> "$FILTERED_LOG"
        done
    else
        # При первом запуске просто объединяем файлы
        for log in "${new_logs[@]}"; do
            cat "$log" >> "$FILTERED_LOG"
        done
    fi

    # Проверяем, появились ли новые строки
    if [ ! -s "$FILTERED_LOG" ]; then
        return 1
    fi
    return 0
}

# 2. Функция генерации разделов отчёта
generate_report() {
    local start_period
    if [ -f "$MARKER_FILE" ]; then
        start_period=$(date -r "$MARKER_FILE" +"%Y-%m-%d %H:%M:%S")
    else
        start_period="Начало ведения лога"
    fi
    local end_period
    end_period=$(date +"%Y-%m-%d %H:%M:%S")

    {
        echo "=== ОТЧЁТ ПО ВЕБ-ТРАФИКУ И ОШИБКАМ ==="
        echo "Период: с [$start_period] по [$end_period]"
        echo "=================================================="
        echo ""
        
        echo "--- 1. IP-адреса с наибольшим числом запросов ---"
        awk '{print $1}' "$FILTERED_LOG" | sort | uniq -c | sort -nr | head -n 15
        echo ""
        
        echo "--- 2. Запрашиваемые URL с наибольшим числом запросов ---"
        awk '{print $7}' "$FILTERED_LOG" | sort | uniq -c | sort -nr | head -n 15
        echo ""
        
        echo "--- 3. Ошибки веб-сервера / приложения (4xx и 5xx коды) ---"
        # Выбираем строки, где HTTP-код начинается с 4 или 5
        awk -F'"' '{print $3}' "$FILTERED_LOG" | awk '$1 ~ /^[45]/ {print $0}' | sort | uniq -c | sort -nr | head -n 20
        if [ ${PIPESTATUS[1]} -eq 0 ] && [ $(awk -F'"' '{print $3}' "$FILTERED_LOG" | awk '$1 ~ /^[45]/ {print $1}' | wc -l) -eq 0 ]; then
            echo "Ошибок не обнаружено."
        fi
        echo ""
        
        echo "--- 4. HTTP-коды ответов и их количество ---"
        awk -F'"' '{print $3}' "$FILTERED_LOG" | awk '{print $1}' | sort | uniq -c | sort -nr
        
    } > "$REPORT_FILE"
}

# ==========================================
# ТОЧКА ВХОДА
# ==========================================
main() {
    # Сбор и фильтрация логов
    if ! filter_new_logs; then
        # Если обновлений в логах нет — отчёт не шлём, просто выходим
        exit 0
    fi

    # Генерация документа
    generate_report

    # Отправка на почту
    mail -s "Веб-отчёт с момента последнего запуска ($(date +'%H:%M'))" "$TO_EMAIL" < "$REPORT_FILE"
}

main
```
#### Добавление скрипта в расписание  cron:
```
0 * * * * /bin/bash /path/to/cron_report.sh > /dev/null 2>&1
```
